📜 Mã SQL khởi tạo bảng (nếu cần xem)
Copy dán vào SQL Editor trên Supabase: 
-- 1. Tạo bảng và mở rộng cột user_id
create table if not exists savings_entries (
  id uuid primary key default gen_random_uuid(),
  user_id uuid default auth.uid(),
  entry_date date not null,
  amount bigint not null,
  note text default 'Thu nhập chạy app',
  created_at timestamptz default now()
);

-- 2. Thêm cột user_id nếu bảng đã có sẵn
alter table savings_entries add column if not exists user_id uuid default auth.uid();
alter table savings_entries drop constraint if exists savings_entries_entry_date_key;

-- 3. Bật phân quyền bảo mật RLS cá nhân (Chỉ chính chủ mới thấy/sửa)
alter table savings_entries enable row level security;
drop policy if exists "Public Access" on savings_entries;
drop policy if exists "Users access own entries" on savings_entries;

create policy "Users access own entries"
  on savings_entries for all
  using (auth.uid() = user_id)
  with check (auth.uid() = user_id);
